## Login in host machine, install first if dont have yet
brew install --cask google-cloud-sdk

# login 
gcloud auth application-default login

# choose project
gcloud auth application-default set-quota-project YOUR_PROJECT_ID 

## run this command to grant Permission 
echo -e "AIRFLOW_UID=$(id -u)" > .env

## Build custom image and Run
docker compose up --build

docker exec -it project4-dbtmodelingforfrauddetectionmetrics-postgres-1 psql -U ecommerce_db

## Check test connection
docker exec -it project4-dbtmodelingforfrauddetectionmetrics-airflow-webserver-1 /bin/bash -c "cd /opt/airflow/ecommerce_db && dbt debug --profiles-dir ."

## After that add packages.yml to dbt_fraud, then run this command to install
docker exec -it project4-dbtmodelingforfrauddetectionmetrics-airflow-webserver-1 /bin/bash -c "cd /opt/airflow/ecommerce_db && dbt deps --profiles-dir ."
